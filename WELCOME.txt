🎯 FEATURES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

 📊 Real-time CPU monitoring (overall + per-core usage)
 💾 Memory tracking (RAM + Swap)
 💿 Disk usage (all partitions)
 🌐 Network traffic (upload/download speeds)
 ⚙️  Process management (top 15 processes)
 🖥️  System information (OS, uptime, kernel)
 🔄 Auto-updates every 1 second via WebSocket
 🎨 Beautiful dark theme with TailwindCSS

⚡ QUICK START
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

 Option 1: Automated Setup
 ──────────────────────────
   .\setup.ps1

 Option 2: Manual Setup
 ──────────────────────
   mix deps.get
   mix assets.setup
   mix assets.build
   mix phx.server

 Then open: http://localhost:4000
